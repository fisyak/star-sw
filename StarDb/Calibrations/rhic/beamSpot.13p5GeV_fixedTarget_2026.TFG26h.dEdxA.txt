TDataSet *CreateTable() {
  if (!gROOT->GetClass("St_beamSpot")) return 0;
  beamSpot_st row;
  St_beamSpot *tableSet = new St_beamSpot("beamSpot",1);
  memset(&row,0,tableSet->GetRowSize());
  row.X =     -0.264; row.sigma_X =      0.274;
  row.Y =     -2.302; row.sigma_Y =      0.193;
  row.Z =    199.448; row.sigma_Z =      0.269;
  tableSet->AddAt(&row);// PVxyz13p5GeV_fixedTarget_2026.TFG26h.dEdxA.root
  return (TDataSet *)tableSet;
}
