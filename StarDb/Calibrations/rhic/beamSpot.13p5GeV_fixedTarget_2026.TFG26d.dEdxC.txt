TDataSet *CreateTable() {
  if (!gROOT->GetClass("St_beamSpot")) return 0;
  beamSpot_st row;
  St_beamSpot *tableSet = new St_beamSpot("beamSpot",1);
  memset(&row,0,tableSet->GetRowSize());
  row.X =     -0.275; row.sigma_X =      0.170;
  row.Y =     -2.218; row.sigma_Y =      0.112;
  row.Z =    200.555; row.sigma_Z =      0.176;
  tableSet->AddAt(&row);// PVxyz13p5GeV_fixedTarget_2026.TFG26d.dEdxC.root
  return (TDataSet *)tableSet;
}
