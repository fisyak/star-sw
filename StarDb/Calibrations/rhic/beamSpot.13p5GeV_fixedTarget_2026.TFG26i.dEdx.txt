TDataSet *CreateTable() {
  if (!gROOT->GetClass("St_beamSpot")) return 0;
  beamSpot_st row;
  St_beamSpot *tableSet = new St_beamSpot("beamSpot",1);
  memset(&row,0,tableSet->GetRowSize());
  row.X =     -0.266; row.sigma_X =      0.280;
  row.Y =     -2.304; row.sigma_Y =      0.200;
  row.Z =    199.457; row.sigma_Z =      0.284;
  tableSet->AddAt(&row);// PVxyz13p5GeV_fixedTarget_2026.TFG26i.dEdx.root
  return (TDataSet *)tableSet;
}
